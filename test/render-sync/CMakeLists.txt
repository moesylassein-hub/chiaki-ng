cmake_minimum_required(VERSION 3.16)
project(chiaki_render_sync_tests LANGUAGES CXX)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(render-sync-unit main.cpp)
target_compile_features(render-sync-unit PRIVATE cxx_std_17)
target_include_directories(render-sync-unit PRIVATE ../../gui/include)
target_link_libraries(render-sync-unit PRIVATE Qt6::Core)
add_test(NAME render-sync COMMAND render-sync-unit)
set_tests_properties(render-sync PROPERTIES TIMEOUT 15)
